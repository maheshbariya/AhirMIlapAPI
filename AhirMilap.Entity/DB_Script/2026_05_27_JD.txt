ALTER TABLE States
ADD StateNameGu NVARCHAR(200)
GO 

ALTER TABLE Districts
ADD DistrictNameGu NVARCHAR(200)
GO 

ALTER  PROCEDURE [dbo].[usp_GetStates]
@Lang NVARCHAR(10) = ''
AS
BEGIN
	SET NOCOUNT ON;

	SELECT Id, 
	 CASE WHEN @Lang = 'gu'
	THEN StateNameGu 
	 ELSE StateName END AS Name
	FROM States 
	WHERE IsActive = 1  
	 ORDER BY
	 CASE WHEN @Lang = 'gu'
	  THEN StateNameGu 
	 ELSE StateName 
	END
END

GO 

ALTER   PROCEDURE [dbo].[usp_GetDistrictByStateId]
@StateId int,
@Lang NVARCHAR(10) = ''
AS
BEGIN
	SET NOCOUNT ON;

	SELECT Id, 
	 CASE WHEN @Lang = 'gu'
	THEN DistrictNameGu
	 ELSE DistrictName END AS Name
	FROM Districts 
	WHERE StateId = @StateId
	ORDER BY 
	 CASE WHEN @Lang = 'gu'
	THEN DistrictNameGu
	 ELSE DistrictName
	END
END


GO

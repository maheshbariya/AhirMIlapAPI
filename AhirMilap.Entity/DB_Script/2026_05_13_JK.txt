CREATE OR ALTER PROCEDURE [dbo].[usp_AddFavoriteUser]
@ProfileIdFrom uniqueidentifier,
@ProfileIdTo uniqueidentifier,
@CreatedOn datetime,
@CreatedBy nvarchar(450)
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @Id UNIQUEIDENTIFIER;

    -- Get existing favorite Id
    SELECT @Id = Id
    FROM FavoriteUser
    WHERE ProfileIdFrom = @ProfileIdFrom
      AND ProfileIdTo = @ProfileIdTo;

    -- If exists then delete
    IF (@Id IS NOT NULL)
    BEGIN
        DELETE FROM FavoriteUser WHERE Id = @Id;
        SELECT 'deleted' AS CreatedBy
        RETURN;
    END

    -- Insert new favorite
    SET @Id = NEWID();

    INSERT INTO FavoriteUser
    (
        Id,
        ProfileIdFrom,
        ProfileIdTo,
        CreatedBy,
        CreatedOn
    )
    VALUES
    (
        @Id,
        @ProfileIdFrom,
        @ProfileIdTo,
        @CreatedBy,
        @CreatedOn
    );

    -- Return inserted record
    SELECT * FROM FavoriteUser WHERE Id = @Id;
END
GO
CREATE OR ALTER PROCEDURE usp_DeleteFavoriteUserById
@Id UNIQUEIDENTIFIER
AS
BEGIN
    SET NOCOUNT ON;

    DELETE FROM FavoriteUser
    WHERE Id = @Id;

    SELECT 1 AS Success;
END
GO
CREATE or alter FUNCTION dbo.GetFullFilePath
(
    @RelativePath NVARCHAR(MAX)
)
RETURNS NVARCHAR(MAX)
AS
BEGIN
    DECLARE @BaseUrl NVARCHAR(500)
    DECLARE @FullPath NVARCHAR(MAX)

    SET @BaseUrl = 'https://transporters033-001-site4.anytempurl.com/'

    -- Return empty if path is null
    IF @RelativePath IS NULL OR LTRIM(RTRIM(@RelativePath)) = ''
        RETURN ''

    -- Replace backslashes with forward slashes
    SET @RelativePath = REPLACE(@RelativePath, '\', '/')

    -- Create full URL
    SET @FullPath = @BaseUrl + @RelativePath

    RETURN @FullPath
END
GO
CREATE OR ALTER PROCEDURE usp_GetFavoriteUsers
@ProfileFromId uniqueidentifier,
@UserId NVARCHAR(450)
AS
/*
usp_GetFavoriteUsers '6EA99C8C-BC07-4DBF-9A0F-3E4562E3D8B9','75CFA3DC-EF7D-4F68-AF6B-1E38F41A13CA'
*/
BEGIN
    SET NOCOUNT ON;

    SELECT F.Id,F.ProfileIdTo,CONCAT(M.FirstName,' ',M.LastName) AS FullName,
	dbo.GetFullFilePath(M.ProfileFilePath) AS ProfileFilePath,M.Occupation,F.CreatedOn
    FROM FavoriteUser F
	INNER JOIN MatrimonyProfiles M ON F.ProfileIdTo = M.Id
    WHERE F.ProfileIdFrom = @ProfileFromId
	ORDER BY CreatedOn DESC
END
 GO

CREATE or alter PROCEDURE usp_SaveNotifications
(
	@Id UNIQUEIDENTIFIER = NULL,
    @SenderProfileId uniqueidentifier,
    @ReceiverProfileId uniqueidentifier,
    @NotificationType NVARCHAR(50),
    @Title NVARCHAR(100),
    @Message NVARCHAR(500),
    @CreatedBy nvarchar(450),
	@CreatedOn datetime
)
AS
BEGIN

    SET NOCOUNT ON;

	-- UPDATE
    IF @Id IS NOT NULL
    BEGIN
        UPDATE Notifications
        SET IsRead = 1,
		UpdatedBy = @CreatedBy,
		UpdatedOn = @CreatedOn
        OUTPUT INSERTED.*
        WHERE Id = @Id;

        RETURN;
    END

    INSERT INTO Notifications
    (
        SenderProfileId,
        ReceiverProfileId,
        NotificationType,
        Title,
        Message, 
        CreatedBy,
		CreatedOn
    )
	OUTPUT INSERTED.*   -- returns newly inserted record
    VALUES
    (
        @SenderProfileId,
        @ReceiverProfileId,
        @NotificationType,
        @Title,
        @Message,
		@CreatedBy,
		@CreatedOn
    ) 
END
GO
CREATE or alter PROCEDURE usp_GetNotificationsByProfileId
(
    @ProfileId uniqueidentifier,
	@PageIndex INT = 1,  
    @PageSize INT = 50  
)
AS
BEGIN

    SET NOCOUNT ON;

    SELECT
        N.Id,
        N.NotificationType,
        N.Title,
        N.Message, 
        N.IsRead,
        N.CreatedOn,
        MP.Id AS SenderProfileId,
        MP.FirstName + ' ' + MP.LastName AS SenderName,
        dbo.GetFullFilePath(MP.ProfileFilePath) AS ProfileFilePath
    FROM Notifications N
    INNER JOIN MatrimonyProfiles MP ON N.SenderProfileId = MP.Id
    WHERE N.ReceiverProfileId = @ProfileId 
    ORDER BY N.CreatedOn DESC
	OFFSET (@PageIndex - 1) * @PageSize ROWS  
    FETCH NEXT @PageSize ROWS ONLY;  
END
GO
CREATE OR ALTER PROCEDURE [dbo].[USP_GetMyChatList]
(
    @ProfileId UNIQUEIDENTIFIER,
    @PageIndex INT = 1,
    @PageSize INT = 50,
    @Search NVARCHAR(100) = NULL
)
AS
-- [dbo].[USP_GetMyChatList] '6EA99C8C-BC07-4DBF-9A0F-3E4562E3D8B9'
BEGIN
    SET NOCOUNT ON;

    ;WITH LastMessages AS
    (
        SELECT 
            m.ChatRoomId,
            m.MessageText,
            m.CreatedOn,
            m.IsRead,
            m.SenderProfileId,

            ROW_NUMBER() OVER
            (
                PARTITION BY m.ChatRoomId
                ORDER BY m.CreatedOn DESC
            ) AS RN

        FROM Messages m
    )

    SELECT
        cr.Id AS ChatRoomId,

        CASE
            WHEN cr.Profile1Id = @ProfileId 
                THEN cr.Profile2Id
            ELSE cr.Profile1Id
        END AS OppositeProfileId,

        mp.FirstName +' '+mp.LastName AS FullName,
        dbo.GetFullFilePath(mp.ProfileFilePath) AS ProfileFilePath,

        lm.MessageText AS LastMessage,
        lm.CreatedOn AS LastMessageTime,

        (
            SELECT COUNT(*)
            FROM Messages um
            WHERE um.ChatRoomId = cr.Id
              AND um.IsRead = 0
              AND um.SenderProfileId != @ProfileId
        ) AS UnreadCount

    FROM ChatRooms cr

    INNER JOIN MatrimonyProfiles mp
        ON mp.Id = CASE
                        WHEN cr.Profile1Id = @ProfileId
                            THEN cr.Profile2Id
                        ELSE cr.Profile1Id
                   END

    LEFT JOIN LastMessages lm
        ON cr.Id = lm.ChatRoomId
       AND lm.RN = 1

    WHERE
        cr.Profile1Id = @ProfileId
        OR cr.Profile2Id = @ProfileId

        AND (
            @Search IS NULL
            OR mp.FirstName +' '+mp.LastName LIKE '%' + @Search + '%'
        )

    ORDER BY lm.CreatedOn DESC

    OFFSET (@PageIndex - 1) * @PageSize ROWS
    FETCH NEXT @PageSize ROWS ONLY;
END
GO

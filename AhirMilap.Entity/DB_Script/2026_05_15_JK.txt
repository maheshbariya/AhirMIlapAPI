CREATE or alter PROCEDURE [dbo].[usp_AddFirebaseToken]
(
    @UserId UNIQUEIDENTIFIER,
    @FcmToken NVARCHAR(500),
	@IsLogOut BIT,
    @CreatedBy NVARCHAR(450),
    @CreatedOn DATETIME
)
AS
BEGIN
    SET NOCOUNT ON;

    IF EXISTS
    (
        SELECT 1
        FROM FirebaseToken
        WHERE UserId = @UserId
        AND FcmToken = @FcmToken
    )
    BEGIN 
        IF (@IsLogOut = 1)
        BEGIN

            DELETE FROM FirebaseToken
            WHERE UserId = @UserId
            AND FcmToken = @FcmToken;

            SELECT 'deleted' AS CreatedBy;
            RETURN;
        END
        ELSE
        BEGIN
            SELECT 'exists' AS CreatedBy;
            RETURN;
        END

    END
    ELSE
    BEGIN 
        INSERT INTO FirebaseToken
        (
            UserId,
            FcmToken,
            CreatedBy,
            CreatedOn
        )
        VALUES
        (
            @UserId,
            @FcmToken,
            @CreatedBy,
            @CreatedOn
        ); 

        SELECT *
        FROM FirebaseToken
        WHERE UserId = @UserId
        AND FcmToken = @FcmToken;
    END
END
GO

CREATE OR ALTER PROCEDURE [dbo].[usp_GetFirebaseTokenByUserId]
(
    @ProfileId uniqueidentifier,
	@LoginUserId NVARCHAR(450)
)
AS
BEGIN
    SET NOCOUNT ON;

	DECLARE @UserId nvarchar(450);
	SELECT @UserId = UserId FROM MatrimonyProfiles WHERE Id = @ProfileId

    SELECT
        Id,
        UserId,
        FcmToken,
        CreatedBy,
        CreatedOn
    FROM FirebaseToken
    WHERE UserId = @UserId
    ORDER BY CreatedOn DESC;
END
GO
